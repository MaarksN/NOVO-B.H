    // ===================================
    // Lógica Principal do Birth Hub 360
    // ===================================
    const views = document.querySelectorAll('.view');
    const backButtonContainer = document.getElementById('back-button-container');
    const viewHistory = [];

    // Por padrão, o tema é escuro, como no site de referência
    document.documentElement.setAttribute('data-theme', 'dark');

    function showView(viewId, isBack = false) {
        const currentView = document.querySelector('.view.active');
        if (currentView) {
            if (!isBack) viewHistory.push(currentView.id);
            currentView.classList.remove('active');
        }
        
        views.forEach(v => v.style.display = 'none');
        const nextView = document.getElementById(viewId);
        if (nextView) {
            nextView.style.display = 'block';
            nextView.classList.add('active');
        }

        backButtonContainer.style.display = viewHistory.length > 0 ? 'block' : 'none';
        lucide.createIcons();
        AOS.refreshHard();
    }
    
    function goBack() {
        if (viewHistory.length > 0) {
            const previousViewId = viewHistory.pop();
            showView(previousViewId, true);
        }
    }

    document.querySelectorAll('a[onclick^="showView"]').forEach(link => {
        link.addEventListener('click', (e) => {
            e.preventDefault();
            const onclickAttr = link.getAttribute('onclick');
            const viewId = onclickAttr.match(/'([^']+)'/)[1];
            showView(viewId);
        });
    });

    // BDR Logic
    const bdrGenerateBtn = document.getElementById('bdr-generate-btn');
    if (bdrGenerateBtn) {
        bdrGenerateBtn.addEventListener('click', () => {
            document.getElementById('bdr-input-section').style.display = 'none';
            document.getElementById('bdr-loading-section').style.display = 'block';
            document.getElementById('bdr-results-section').style.display = 'none';
            setTimeout(() => {
                document.getElementById('bdr-loading-section').style.display = 'none';
                const resultsContainer = document.getElementById('bdr-results-section');
                resultsContainer.style.display = 'block';
                resultsContainer.querySelector('.space-y-8').innerHTML = `

                    <div class="card p-6" data-aos="fade-up" data-aos-delay="300"><h3 class="text-xl font-bold mb-4">Plano de Ação e Execução no CRM</h3></div>

                AOS.refreshHard();

<script>
    AOS.init({ duration: 800, once: true });
    lucide.createIcons();

    const apiKey = "";
    const geminiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-09-2025:generateContent?key=${apiKey}`;
    const ttsUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-tts:generateContent?key=${apiKey}`;

    // Utility: Exponential Backoff Fetch
    async function fetchWithRetry(url, options, retries = 5, delay = 1000) {
        for (let i = 0; i < retries; i++) {
            try {
                const response = await fetch(url, options);
                if (response.ok) return await response.json();
            } catch (e) {
                if (i === retries - 1) throw e;
            }
            await new Promise(r => setTimeout(r, delay));
            delay *= 2;
        }
    }

    // Logic: Call Gemini Text
    async function callGemini(prompt, btnId, outId) {
        const btn = document.getElementById(btnId);
        const out = document.getElementById(outId);
        const originalText = btn.innerHTML;

        btn.disabled = true;
        btn.innerHTML = `<i class="fas fa-circle-notch fa-spin mr-2"></i> IA Processando...`;
        out.classList.remove('hidden');
        out.innerHTML = "O Gemini está desenhando a estratégia...";

        try {
            const data = await fetchWithRetry(geminiUrl, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({
                    contents: [{ parts: [{ text: prompt }] }],
                    generationConfig: { temperature: 0.7 }
                })
            });
            const text = data.candidates?.[0]?.content?.parts?.[0]?.text || "Falha na resposta.";
            out.innerText = text;
        } catch (err) {
            out.innerText = "Erro na conexão com o cérebro da IA. Tente novamente.";
        } finally {
            btn.innerHTML = originalText;
            btn.disabled = false;
        }
    }

    // Feature: Text to Speech (Voz do Cliente)
    async function playObjectionVoice() {
        const text = document.getElementById('ae-objection').value;
        if(!text) return;
        const btn = document.getElementById('ae-audio-btn');
        const original = btn.innerHTML;
        btn.innerHTML = `<i class="fas fa-spinner fa-spin mr-2"></i> Gerando Voz...`;

        try {
            const response = await fetch(ttsUrl, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({
                    contents: [{ parts: [{ text: `Diga esta frase com um tom de cliente cético ou levemente irritado: ${text}` }] }],
                    generationConfig: {
                        responseModalities: ["AUDIO"],
                        speechConfig: { voiceConfig: { prebuiltVoiceConfig: { voiceName: "Puck" } } }
                    },
                    model: "gemini-2.5-flash-preview-tts"
                })
            });
            const result = await response.json();
            const pcmData = result.candidates[0].content.parts[0].inlineData.data;
            
            // Conversão PCM para WAV baseada na especificação do ambiente
            const binaryString = atob(pcmData);
            const len = binaryString.length;
            const bytes = new Uint8Array(len);
            for (let i = 0; i < len; i++) bytes[i] = binaryString.charCodeAt(i);
            const blob = new Blob([bytes], { type: 'audio/wav' });
            const url = URL.createObjectURL(blob);
            new Audio(url).play();
        } catch (e) {
            console.error("TTS Error:", e);
        } finally {
            btn.innerHTML = original;
        }
    }

            const page = target.dataset.page;
            if (route) showView(route);
            if (page) showAppPage(page);

                    e.preventDefault();

                    if (draggable) column.appendChild(draggable);

        // Função Genérica para API Gemini
         async function callGeminiAPI(prompt, isJson = false) {
            let chatHistory = [{ role: "user", parts: [{ text: prompt }] }];
            const payload = { contents: chatHistory };
            
            if (isJson) {
                payload.generationConfig = {
                    responseMimeType: "application/json",
                };
            }

            const apiKey = ""; // A chave é injetada pelo ambiente
            const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-05-20:generateContent?key=${apiKey}`;
            
            const response = await fetch(apiUrl, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify(payload)
            });

            if (!response.ok) throw new Error(`API Error: ${response.statusText}`);
            const result = await response.json();
            
            if (result.candidates && result.candidates[0]?.content?.parts[0]?.text) {
                return result.candidates[0].content.parts[0].text;
            } else {
                throw new Error("A resposta da API não contém o texto esperado.");
            }
        }
